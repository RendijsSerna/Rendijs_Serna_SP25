--Task 1. Figure out what security precautions are already used in your 'dvd_rental' database.  Prepare description
-- 15 different roles and user postgres
-- only role that inherits other roles 
CREATE ROLE pg_monitor WITH 
	NOSUPERUSER
	NOCREATEDB
	NOCREATEROLE
	INHERIT
	NOLOGIN
	NOREPLICATION
	NOBYPASSRLS
	CONNECTION LIMIT -1;

GRANT pg_read_all_settings TO pg_monitor;
GRANT pg_read_all_stats TO pg_monitor;
GRANT pg_stat_scan_tables TO pg_monitor;
